      - db

      - DB_HOST=db

  db:
    image: mysql:8
    container_name: db  # <--- Esto es vital para que conexion.php lo encuentre
    environment:
      MYSQL_ROOT_PASSWORD: root123
      MYSQL_DATABASE: verde_vida
    volumes:
      - db_data:/var/lib/mysql
      - ./db_init:/docker-entrypoint-initdb.d

  phpmyadmin:
    image: phpmyadmin/phpmyadmin

      - "8081:80"

Para el Dockerfile

FROM php:8.2-apache

# Instalar la extensión mysqli que tus scripts necesitan
RUN docker-php-ext-install mysqli && docker-php-ext-enable mysqli

# Copiar el código al contenedor
COPY . /var/www/html/

# Dar permisos a la carpeta de servidor
RUN chown -R www-data:www-data /var/www/html

Para crear la base de datos

mkdir db_init && cd db_init && nano init.sql

-- Crear la base de datos si no existe
CREATE DATABASE IF NOT EXISTS verde_vida;
USE verde_vida;

-- 1. Tabla de Usuarios (para login y registro de personal)
CREATE TABLE IF NOT EXISTS usuarios (
    ID INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    apellido VARCHAR(100) NOT NULL,
    email VARCHAR(100) NOT NULL UNIQUE,
    direccion VARCHAR(255),
    telefono VARCHAR(20),
    contraseña VARCHAR(255) NOT NULL, -- El script PHP usa 'contraseña'
    rol VARCHAR(20) DEFAULT 'jardinero',
    fecha_contratacion DATE,
    hora DATETIME
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 2. Tabla de Especies (información botánica general)
CREATE TABLE IF NOT EXISTS especies (
    ID INT AUTO_INCREMENT PRIMARY KEY,
    nombre_comun VARCHAR(100) NOT NULL,
    nombre_cientifico VARCHAR(100),
    familia VARCHAR(100),
    origen VARCHAR(100),
    tipo_planta VARCHAR(50),
    descripcion TEXT,
    dificultad_cultivo VARCHAR(50),
    hora DATETIME
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 3. Tabla de Inventario (stock físico relacionado a una especie)
CREATE TABLE IF NOT EXISTS inventario (
    ID INT AUTO_INCREMENT PRIMARY KEY,
    ID_especie INT NOT NULL,
    cantidad_disponible INT DEFAULT 0,
    precio_venta DECIMAL(10, 2),
    precio_costo DECIMAL(10, 2),
    estado VARCHAR(50),
    calidad VARCHAR(50),
    fecha_ingreso DATE,
    ubicacion VARCHAR(100),
    notas TEXT,
    hora DATETIME,
    CONSTRAINT fk_especie FOREIGN KEY (ID_especie) 
        REFERENCES especies(ID) 
        ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
